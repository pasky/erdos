#!/usr/bin/env bash
set -uo pipefail

# Clean committed-source gate for L17, L20, L21 and L22
# ---------------------------------------------------------------------------
# Runs the committed Lean gate in clean mode against the checked source, the
# tree as it stood on 2026-10-13T04:56:43Z (after the restore), and
# captures both streams, the UTC window, the worktree state and
# the exit code beside this script in clean_gate_log.txt. The gate archives
# that source's lean/ tree into a fresh temporary directory, fetches the
# Mathlib cache, builds, runs the axiom audit against the committed manifest
# and checks the self-test stamp. The runner is a non-author, not the
# restorer. One run serves L17, L20, L21 and L22, the four claims in that
# source's manifest.
#
# The filed clean_gate_log.txt has the commit id its two worktree_head lines
# printed replaced by the words "the checked source"; the streams printed no
# detail that identifies the computer, so no other line of the log changed.

# REV, this script's one argument, names the checked source; the record
# restates it by the date it stood on.
REV="$1"
SOURCE="the tree as it stood on 2026-10-13T04:56:43Z, before it landed on the default branch"
HERE="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd -P)"
REPO="$(git -C "$HERE" rev-parse --show-toplevel)"
LOG="$HERE/clean_gate_log.txt"

# Cap Lake's parallel compiles, as the gate's own default does.
export LEAN_NUM_THREADS=4

cd "$REPO"
START_UTC="$(date -u +%Y-%m-%dT%H:%M:%SZ)"
START_S="$(date +%s)"
{
    echo "clean gate of the archived lean/ subtree, non-author runner, fresh context"
    echo "checked_source: $SOURCE"
    echo "lean_tree: lean/ as it stood on 2026-10-13T04:56:43Z"
    echo "command: LEAN_NUM_THREADS=4 nice -n 10 /bin/bash lean/scripts/gate.sh --clean, run on $SOURCE"
    echo "worktree_head_before: $(git rev-parse HEAD)"
    echo "worktree_status_before: $(git status --porcelain --untracked-files=no | wc -l | tr -d ' ') tracked changes"
    echo "start_utc: $START_UTC"
    echo "--- streams (stdout and stderr interleaved) ---"
} > "$LOG"
# Run the gate at lowered scheduling priority beside other work.
nice -n 10 /bin/bash lean/scripts/gate.sh --clean "$REV" >> "$LOG" 2>&1
RC=$?
END_UTC="$(date -u +%Y-%m-%dT%H:%M:%SZ)"
END_S="$(date +%s)"
{
    echo "--- end streams ---"
    echo "worktree_head_after: $(git rev-parse HEAD)"
    echo "worktree_status_after: $(git status --porcelain --untracked-files=no | wc -l | tr -d ' ') tracked changes"
    echo "end_utc: $END_UTC"
    echo "wall_seconds: $((END_S - START_S))"
    echo "exit_code: $RC"
} >> "$LOG"
exit "$RC"
